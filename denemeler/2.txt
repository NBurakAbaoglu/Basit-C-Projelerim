        function addRow() {
            const tbody = document.querySelector('.excel-table tbody');
            const newRow = document.createElement('tr');
            
            // Satır başlığı için boş hücre
            const headerCell = document.createElement('td');
            headerCell.className = 'row-header';
            newRow.appendChild(headerCell);
            
            // Boş hücreler
            for (let i = 0; i < 5; i++) {
                const cell = document.createElement('td');
                cell.className = 'empty-cell';
                newRow.appendChild(cell);
            }
            
            tbody.appendChild(newRow);
        }
        
                 function addColumn() {
             const table = document.querySelector('.excel-table');
             const rows = table.querySelectorAll('tr');
             
             rows.forEach((row, index) => {
                 const cell = document.createElement(index === 0 ? 'th' : 'td');
                 if (index === 0) {
                     cell.className = 'empty-cell';
                 } else {
                     cell.className = 'empty-cell';
                 }
                 row.appendChild(cell);
             });
         }
         
                   function moveOrganizationHeader(headerElement) {
              const currentTh = headerElement.parentElement;
              const nextTh = currentTh.nextElementSibling;
              
                                 if (nextTh && !nextTh.classList.contains('corner-cell')) {
                       // Mevcut hücreyi düzenlenebilir yap
                       currentTh.innerHTML = '<textarea class="editable-cell" placeholder="Organizasyon adı yazın..." onblur="createOrganizationButton(this)" onkeydown="handleKeyDown(event, this)" oninput="resizeTextarea(this)"></textarea>';
                       currentTh.querySelector('textarea').focus();
                  
                  // Başlığı bir sonraki sütuna taşı
                  nextTh.innerHTML = '<div class="rotated-header" onclick="moveOrganizationHeader(this)">Organizasyon Ekle</div>';
              } else {
                  // Eğer sonraki sütun yoksa, yeni sütun ekle
                  addColumn();
                  
                  // Yeni eklenen sütunu bul
                  const newNextTh = currentTh.nextElementSibling;
                  if (newNextTh) {
                      // Mevcut hücreyi düzenlenebilir yap
                      currentTh.innerHTML = '<textarea class="editable-cell" placeholder="Organizasyon adı yazın..." onblur="createOrganizationButton(this)" onkeydown="handleKeyDown(event, this)" oninput="resizeTextarea(this)"></textarea>';
                      currentTh.querySelector('textarea').focus();
                      
                      // Başlığı yeni sütuna taşı
                      newNextTh.innerHTML = '<div class="rotated-header" onclick="moveOrganizationHeader(this)">Organizasyon Ekle</div>';
                  }
              }
          }

                                                                                                                                                                                  function createOrganizationButton(inputElement) {
                const text = inputElement.value.trim();
                if (text) {
                    const th = inputElement.parentElement;
                    
                    // Metin boyutuna göre yükseklik ve genişlik hesapla
                    const lines = text.split('\n');
                    const minHeight = 120;
                    const minWidth = 30;
                    const heightPerChar = 12; // Her karakter için yükseklik
                    const widthPerLine = 15; // Her satır için genişlik
                    
                    // Yükseklik hesaplama - toplam karakter sayısına göre
                    const totalChars = text.length;
                    const calculatedHeight = Math.max(minHeight, totalChars * heightPerChar);
                    
                    // Genişlik hesaplama - satır sayısına göre
                    const calculatedWidth = Math.max(minWidth, Math.min(200, lines.length * widthPerLine));
                    
                    th.innerHTML = `<button class="organization-button" onclick="editOrganization(this)" style="height: ${calculatedHeight}px; width: ${calculatedWidth}px;">${text}</button>`;
                    
                    // Bu sütunun altındaki her satıra yuvarlak daire ekle
                    addDotsToColumn(th);
                } else {
                    // Boşsa "Organizasyon Ekle" butonunu eski yerine taşı
                    const th = inputElement.parentElement;
                    const table = th.closest('table');
                    const thead = table.querySelector('thead tr');
                    
                    // Mevcut hücreyi boş yap
                    th.innerHTML = '';
                    
                    // Tüm "Organizasyon Ekle" butonlarını temizle
                    const allRotatedHeaders = thead.querySelectorAll('.rotated-header');
                    allRotatedHeaders.forEach(header => {
                        if (header.textContent === 'Organizasyon Ekle') {
                            header.parentElement.innerHTML = '';
                        }
                    });
                    
                    // "Organizasyon Ekle" butonunu sadece ilk sütuna (corner'dan sonraki) ekle
                    const firstTh = thead.children[1]; // Corner'dan sonraki ilk sütun
                    if (firstTh) {
                        firstTh.innerHTML = '<div class="rotated-header" onclick="moveOrganizationHeader(this)">Organizasyon Ekle</div>';
                    }
                }
            }
         
                                                                                                                                                                                                                                                                                                                               function editOrganization(buttonElement) {
               const text = buttonElement.textContent;
               const th = buttonElement.parentElement;
               // Mevcut boyutları koru
               const currentHeight = buttonElement.style.height || '120px';
               const currentWidth = buttonElement.style.width || '30px';
               th.innerHTML = `<textarea class="editable-cell" onblur="createOrganizationButton(this)" onkeydown="handleKeyDown(event, this)" oninput="resizeTextarea(this)" style="height: ${currentHeight}; width: ${currentWidth};">${text}</textarea>`;
               th.querySelector('textarea').focus();
               
                               // Textarea'yı otomatik olarak boyutlandır
                setTimeout(() => {
                    const textarea = th.querySelector('textarea');
                    if (textarea) {
                        const lines = textarea.value.split('\n');
                        const totalChars = textarea.value.length;
                        const newHeight = Math.max(120, totalChars * 12);
                        const newWidth = Math.max(30, Math.min(200, lines.length * 15));
                        textarea.style.height = newHeight + 'px';
                        textarea.style.width = newWidth + 'px';
                    }
                }, 0);
                      }
           
                                                                                                                                                                                               function handleKeyDown(event, textarea) {
                   if (event.key === 'Enter') {
                       // Enter tuşuna basıldığında yeni satır ekle
                       const start = textarea.selectionStart;
                       const end = textarea.selectionEnd;
                       const value = textarea.value;
                       
                       textarea.value = value.substring(0, start) + '\n' + value.substring(end);
                       textarea.selectionStart = textarea.selectionEnd = start + 1;
                       
                       // Textarea boyutunu otomatik ayarla
                       setTimeout(() => {
                           resizeTextarea(textarea);
                       }, 0);
                       
                       event.preventDefault();
                   }
               }
               
                                                                                                                                                                                                                                                               // Textarea boyutunu otomatik ayarlayan fonksiyon
                 function resizeTextarea(textarea) {
                     const lines = textarea.value.split('\n');
                     const totalChars = textarea.value.length;
                     const heightPerChar = 12;
                     const widthPerLine = 15;
                     const newHeight = Math.max(120, totalChars * heightPerChar);
                     const newWidth = Math.max(30, Math.min(200, lines.length * widthPerLine));
                     textarea.style.height = newHeight + 'px';
                     textarea.style.width = newWidth + 'px';
                 }
            
                         function addNameRow(clickedCell) {
                 const currentRow = clickedCell.parentElement;
                 const tbody = currentRow.parentElement;
                 
                 // Eğer tıklanan hücre boşsa, input alanı aç
                 if (!clickedCell.textContent.trim()) {
                     clickedCell.innerHTML = '<input type="text" class="name-input" placeholder="İsim yazın..." onblur="createNameText(this)" onkeydown="handleNameKeyDown(event, this)">';
                     clickedCell.querySelector('input').focus();
                 } else {
                     // Eğer hücrede zaten isim varsa, altına yeni satır ekle
                     const newRow = document.createElement('tr');
                     
                     // Satır başlığı için boş hücre
                     const headerCell = document.createElement('td');
                     headerCell.className = 'row-header';
                     headerCell.onclick = function() { addNameRow(this); };
                     newRow.appendChild(headerCell);
                     
                     // Boş hücreler (sütun sayısına göre)
                     const columnCount = currentRow.children.length - 1;
                     for (let i = 0; i < columnCount; i++) {
                         const cell = document.createElement('td');
                         cell.className = 'empty-cell';
                         newRow.appendChild(cell);
                     }
                     
                     // Yeni satırı mevcut satırın altına ekle
                     tbody.insertBefore(newRow, currentRow.nextSibling);
                     
                     // Yeni eklenen hücreyi input alanına çevir
                     const newHeaderCell = newRow.querySelector('.row-header');
                     newHeaderCell.innerHTML = '<input type="text" class="name-input" placeholder="İsim yazın..." onblur="createNameText(this)" onkeydown="handleNameKeyDown(event, this)">';
                     newHeaderCell.querySelector('input').focus();
                 }
             }
            
                         function createNameText(inputElement) {
                 const text = inputElement.value.trim();
                 if (text) {
                     const td = inputElement.parentElement;
                     td.innerHTML = text;
                     td.onclick = function() { addNameRow(this); };
                     
                     // İsim girildiğinde organizasyon sütunlarına daire ekle
                     addDotsToNameRow(td.parentElement);
                     
                     // İsim girildiğinde otomatik olarak altına boş hücre ekle
                     setTimeout(() => {
                         addNameRow(td);
                     }, 100);
                 } else {
                     // Boşsa boş hücre olarak bırak
                     const td = inputElement.parentElement;
                     td.innerHTML = '';
                     td.onclick = function() { addNameRow(this); };
                 }
             }
            
                         function editName(textElement) {
                 const text = textElement.textContent;
                 const td = textElement.parentElement;
                 td.innerHTML = `<input type="text" class="name-input" value="${text}" onblur="createNameText(this)" onkeydown="handleNameKeyDown(event, this)">`;
                 td.querySelector('input').focus();
             }
            
                         function handleNameKeyDown(event, input) {
                 if (event.key === 'Enter') {
                     input.blur(); // Input'u kapat ve metin oluştur
                     event.preventDefault();
                 }
             }

        function addDotsToColumn(th) {
            const tbody = th.closest('table').querySelector('tbody');
            const rows = tbody.querySelectorAll('tr');
            const columnIndex = th.cellIndex;

            rows.forEach(row => {
                if (row.cells[columnIndex]) {
                    // Sadece isim olan satırlara daire ekle (boş satırlara ekleme)
                    const nameCell = row.querySelector('.row-header');
                    if (nameCell && nameCell.textContent.trim() !== '') {
                        row.cells[columnIndex].innerHTML = '<div class="organization-dot"></div>';
                    }
                }
            });
        }

        function addDotsToNameRow(row) {
            const table = row.closest('table');
            const thead = table.querySelector('thead tr');
            const tbody = table.querySelector('tbody');
            
            // Organizasyon sütunlarını bul (organization-button olan sütunlar)
            for (let i = 1; i < thead.children.length; i++) {
                const headerCell = thead.children[i];
                if (headerCell.querySelector('.organization-button')) {
                    // Bu sütun organizasyon sütunu, daire ekle
                    const cell = row.cells[i];
                    if (cell && !cell.querySelector('.organization-dot')) {
                        cell.innerHTML = '<div class="organization-dot"></div>';
                    }
                }
            }
        }
